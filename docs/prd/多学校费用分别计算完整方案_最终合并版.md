# 多学校费用分别计算完整方案 - 最终合并版

## 文档概述

本文档是针对 One-API 系统的多学校费用分别计算需求的最终完整方案，融合了学科组扩展功能和低耦合设计理念。本方案在保持原有系统完全兼容的基础上，实现了学校 → 学科组 → 老师的三级费用管理体系，并采用扩展日志表设计，确保系统升级的便利性和安全性。

**核心特性**：
1. 完善了用户学校映射表的维护方案
2. 统一了字段命名规范，消除歧义  
3. 设计了老师映射的自动化方案
4. **新增**：支持学科组维度的精细化管理
5. **新增**：采用低耦合的扩展日志表设计
6. **新增**：One-API内部独立维护学科组信息

---

## 1. 业务需求分析

### 1.1 业务背景

**系统架构链路**：
- **课件制作平台**：前端应用，专注于课件制作功能，维护学校和老师基本信息
- **Dify系统**：后台业务逻辑系统，通过工作流调用大模型
- **One-API系统**：大模型聚合中转系统，统一管理多个大模型提供商，独立维护学科组信息
- **用户层级**：学校 → 学科组 → 老师，需要按学科组维度进行费用计算

**当前配置**：
```bash
# Dify系统配置
BASE_URL=http://api.hk33smarter.com/v1
APIKEY=sk-xxx

# One-API系统配置
渠道1：deepseek渠道 → deepseek-chat模型
渠道2：硅基流动 → Qwen2.5-VL-72B模型
```

### 1.2 核心需求

1. **学科组级别费用分离**：每个学科组的大模型使用费用独立计算
2. **老师到学科组映射**：通过 teacher_id 识别老师所属学科组
3. **多模型提供商支持**：每个学科组可采购不同的模型提供商
4. **自动模型选择**：根据学科组采购情况自动选择可用模型
5. **独立配额管理**：学科组间配额完全隔离，互不影响
6. **低耦合设计**：保持原有系统完全不受影响，便于未来升级

### 1.3 技术约束

- Dify 只能通过 header 提供 user_id（实际为老师ID），无学校和学科组信息
- 不能在 One-API 为每个老师创建账号（用户量大）
- 需要保持 One-API 的高性能特性
- 尽量不改动课件制作平台和 Dify 的业务逻辑
- 课件平台不维护学科组信息，学科组完全由One-API内部管理

)

const (
    SubjectStatusActive   = 1

)

type Subject struct {
    Id        int    `json:"id" gorm:"primaryKey"`
    SchoolId  int    `json:"school_id"`
    Name      string `json:"name" gorm:"size:100"`
    Code      string `json:"code" gorm:"size:50;uniqueIndex:idx_school_code"`
    GroupName string `json:"group_name" gorm:"size:50;unique"`
    Description string `json:"description"`
    Status    int    `json:"status" gorm:"default:1"`
    CreatedBy int    `json:"created_by"`
    CreatedAt int64  `json:"created_at"`
    UpdatedAt int64  `json:"updated_at"`
    
    // 关联查询字段

// 缓存获取学科组信息
func CacheGetSubject(ctx context.Context, subjectId int) (*Subject, error) {
    if !common.RedisEnabled {
        return GetSubject(ctx, subjectId)
    }
    
    cacheKey := fmt.Sprintf("subject:%d", subjectId)
    subjectStr, err := common.RedisGet(cacheKey)
    if err != nil {
        subject, err := GetSubject(ctx, subjectId)
        if err != nil {
            return nil, err
        }
        
        subjectJson, _ := json.Marshal(subject)
        _ = common.RedisSet(cacheKey, string(subjectJson), time.Hour)

        // 先执行后续中间件和处理器
        c.Next()
        
        // 检查是否需要记录扩展日志

// 判断是否需要记录扩展日志
func shouldRecordExtendedLog(c *gin.Context) bool {
    // 只有在有学科组信息时才记录扩展日志
    subjectId := c.GetInt(ctxkey.SubjectId)
    teacherId := c.GetString(ctxkey.TeacherId)
    
    return subjectId > 0 || teacherId != ""
}

// 记录扩展日志
func recordExtendedLog(c *gin.Context) {
    ctx := context.Background()
    
    // 获取原日志ID（从响应头或其他方式获取）
    logIdStr := c.GetHeader("X-Log-ID")
    if logIdStr == "" {
        // 如果没有日志ID，可能需要通过其他方式获取最新的日志记录
        logIdStr = getLatestLogId(c)
    }
    
    if logIdStr == "" {
        logger.Errorf(ctx, "无法获取日志ID，跳过扩展日志记录")
        return
    }
    
    logId, err := strconv.Atoi(logIdStr)
    if err != nil {
        logger.Errorf(ctx, "日志ID格式错误: %s", logIdStr)
        return
    }
    
    // 构建扩展日志信息
    extendedInfo := &model.ExtendedLogInfo{
        SchoolId:    c.GetInt(ctxkey.SchoolId),
        SchoolName:  c.GetString(ctxkey.SchoolName),

        TeacherId:   c.GetString(ctxkey.TeacherId),
        TeacherName: c.GetString(ctxkey.TeacherName),
        GroupName:   c.GetString(ctxkey.Group),

type SubjectUsageStats struct {

// 获取学科组使用统计
func GetSubjectUsageStats(ctx context.Context, schoolId int, startTime, endTime int64) ([]SubjectUsageStats, error) {
    var stats []SubjectUsageStats
    
    query := `
        SELECT 
            s.id as subject_id,
            s.name as subject_name,
            s.school_id,
            sc.name as school_name,
            COUNT(l.id) as request_count,
            SUM(l.prompt_tokens + l.completion_tokens) as total_tokens,
            SUM(l.quota) as total_quota,
            COUNT(DISTINCT el.teacher_id) as teacher_count,
            CASE 
                WHEN COUNT(DISTINCT el.teacher_id) > 0 
                THEN SUM(l.quota) / COUNT(DISTINCT el.teacher_id)
                ELSE 0 
            END as avg_quota_per_teacher
        FROM subjects s
        LEFT JOIN schools sc ON s.school_id = sc.id
        LEFT JOIN extended_logs el ON s.id = el.subject_id
        LEFT JOIN logs l ON el.log_id = l.id
        WHERE s.status = 1
    `
    
    args := []interface{}{}
    
    if schoolId > 0 {
        query += " AND s.school_id = ?"
        args = append(args, schoolId)
    }
    
    if startTime > 0 {
        query += " AND l.created_at >= ?"
        args = append(args, startTime)
    }
    
    if endTime > 0 {
        query += " AND l.created_at <= ?"
        args = append(args, endTime)
    }
    
    query += " GROUP BY s.id, s.name, s.school_id, sc.name ORDER BY total_quota DESC"
    
    err := DB.WithContext(ctx).Raw(query, args...).Scan(&stats).Error
    return stats, err
