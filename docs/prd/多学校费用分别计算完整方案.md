# 多学校费用分别计算完整方案 - 专家评审版（优化）

## 文档概述

本文档针对 One-API 系统的多学校费用分别计算需求，提供了完整的技术方案设计和实现细节。本版本已针对专家反馈进行了重要优化：
1. 完善了用户学校映射表的维护方案
2. 统一了字段命名规范，消除歧义
3. 设计了老师映射的自动化方案

---

## 1. 业务需求分析

### 1.1 业务背景

**系统架构链路**：
- **课件制作平台**：前端应用，专注于课件制作功能
- **Dify系统**：后台业务逻辑系统，通过工作流调用大模型
- **One-API系统**：大模型聚合中转系统，统一管理多个大模型提供商
- **用户层级**：学校 → 老师，需要按学校维度进行费用计算

**当前配置**：
```bash
# Dify系统配置
BASE_URL=http://api.hk33smarter.com/v1
APIKEY=sk-xxx

# One-API系统配置
渠道1：deepseek渠道 → deepseek-chat模型
渠道2：硅基流动 → Qwen2.5-VL-72B模型
```

### 1.2 核心需求

1. **学校级别费用分离**：每个学校的大模型使用费用独立计算
2. **老师到学校映射**：通过 teacher_id 识别老师所属学校
3. **多模型提供商支持**：每个学校可采购不同的模型提供商
4. **自动模型选择**：根据学校采购情况自动选择可用模型
5. **独立配额管理**：学校间配额完全隔离，互不影响

### 1.3 技术约束

- Dify 只能通过 header 提供 user_id（实际为老师ID），无学校信息
- 不能在 One-API 为每个老师创建账号（用户量大）
- 需要保持 One-API 的高性能特性
- 尽量不改动课件制作平台和 Dify 的业务逻辑

```go
type OfflineTask struct {

func CreateOfflineTaskForTeacherMapping(teacherId string) error {
    task := &OfflineTask{
        TaskType:  "teacher_mapping",
        TeacherId: teacherId,
        Status:    1, // 待处理
        Retries:   0,
        CreatedAt: time.Now().Unix(),
        UpdatedAt: time.Now().Unix(),
    }
    
    return DB.Create(task).Error
}
```

#### 5.2.2 离线脚本处理
```go
// scripts/teacher_mapping_worker.go
func ProcessTeacherMappingTasks() {
    for {
        // 1. 获取待处理的任务
        var tasks []OfflineTask
        err := DB.Where("task_type = ? AND status = ?", "teacher_mapping", 1).
            Limit(10).Find(&tasks).Error
        if err != nil {
            logger.Error("获取离线任务失败:", err)
            time.Sleep(10 * time.Second)
            continue
        }
        
        // 2. 处理每个任务
        for _, task := range tasks {
            err := processTeacherMappingTask(task)
            if err != nil {
                logger.Errorf("处理任务失败: %v", err)
                // 更新任务状态为失败
                updateTaskStatus(task.Id, 4, task.Retries+1)
            } else {
                // 更新任务状态为完成
                updateTaskStatus(task.Id, 3, task.Retries)
            }
        }
        
        // 3. 等待下次处理
        time.Sleep(30 * time.Second)
    }
}

// 处理单个老师映射任务
func processTeacherMappingTask(task OfflineTask) error {
    // 1. 更新任务状态为处理中
    updateTaskStatus(task.Id, 2, task.Retries)
    
    // 2. 调用课件制作平台API获取老师学校映射
    schoolId, err := getTeacherSchoolFromPlatform(task.TeacherId)
    if err != nil {
        return fmt.Errorf("获取老师学校信息失败: %v", err)
    }
    
    // 3. 创建映射关系
    mapping := &TeacherSchoolMapping{
        TeacherId: task.TeacherId,
        SchoolId:  schoolId,
        Status:    MappingStatusActive,
        CreatedAt: time.Now().Unix(),
        UpdatedAt: time.Now().Unix(),
    }
    
    err = DB.Create(mapping).Error
    if err != nil {
        return fmt.Errorf("创建映射关系失败: %v", err)
    }
    
    // 4. 清除缓存
    ClearTeacherMappingCache(task.TeacherId)
    
    return nil
}

// 调用课件制作平台API
func getTeacherSchoolFromPlatform(teacherId string) (int, error) {
    // 构造请求URL
    url := fmt.Sprintf("%s/api/teacher/%s/school", config.PlatformAPIURL, teacherId)
    
    // 发送HTTP请求
    resp, err := http.Get(url)
    if err != nil {
        return 0, err
    }
    defer resp.Body.Close()
    
    // 解析响应
    var response struct {
        Success bool `json:"success"`
        Data    struct {
            SchoolId int `json:"school_id"`
        } `json:"data"`
    }
    
    err = json.NewDecoder(resp.Body).Decode(&response)
    if err != nil {
        return 0, err
    }
    
    if !response.Success {
        return 0, fmt.Errorf("平台API返回失败")
    }
    
    return response.Data.SchoolId, nil
}
```

### 5.3 动态映射创建（优化版）

#### 5.3.1 首次请求时的处理策略
```go
// middleware/school_auth.go (优化版)
func SchoolAuth() gin.HandlerFunc {
    return func(c *gin.Context) {
        ctx := c.Request.Context()
        
        // 1. 获取老师ID
        teacherId := c.GetHeader("X-User-ID")
        if teacherId == "" {
            c.Next()
            return
        }
        
        // 2. 查询老师映射（带缓存）
        mapping, err := model.CacheGetTeacherSchoolMapping(ctx, teacherId)
        if err != nil {
            // 3. 如果映射不存在，使用默认分组
            if errors.Is(err, gorm.ErrRecordNotFound) {
                logger.Infof(ctx, "老师 %s 映射不存在，使用默认分组", teacherId)
                
                // 3.1 设置默认分组
                c.Set(ctxkey.Group, "default")
                c.Set(ctxkey.SchoolId, 0)
                c.Set(ctxkey.SchoolName, "默认学校")
                c.Set(ctxkey.TeacherId, teacherId)
                
                // 3.2 创建离线任务
                go model.CreateOfflineTaskForTeacherMapping(teacherId)
                
                c.Next()
                return
            } else {
                logger.Errorf(ctx, "获取老师映射失败: %v", err)
                abortWithMessage(c, http.StatusInternalServerError, "系统错误")
                return
            }
        }
        
        // 4. 设置学校上下文
        c.Set(ctxkey.SchoolId, mapping.SchoolId)
        c.Set(ctxkey.SchoolName, mapping.SchoolName)
        c.Set(ctxkey.Group, mapping.GroupName)
        c.Set(ctxkey.TeacherId, teacherId)
        
        logger.Debugf(ctx, "学校认证成功: 老师ID=%s, 学校=%s, 用户组=%s", 
            teacherId, mapping.SchoolName, mapping.GroupName)
        
        c.Next()
    }
}
```

```go
const (
    MappingStatusActive   = 1
    MappingStatusInactive = 2
)

type TeacherSchoolMapping struct {
    TeacherId string `json:"teacher_id" gorm:"primaryKey;size:100"`
    SchoolId  int    `json:"school_id"`
    Status    int    `json:"status" gorm:"default:1"`
    CreatedAt int64  `json:"created_at"`
    UpdatedAt int64  `json:"updated_at"`
    
    // 关联查询用的字段
    SchoolName string `json:"school_name" gorm:"-"`
    GroupName  string `json:"group_name" gorm:"-"`
}

// 缓存获取老师学校映射
func CacheGetTeacherSchoolMapping(ctx context.Context, teacherId string) (*TeacherSchoolMapping, error) {
    if !common.RedisEnabled {
        return GetTeacherSchoolMapping(ctx, teacherId)
    }
    
    cacheKey := fmt.Sprintf("teacher_mapping:%s", teacherId)
    mappingStr, err := common.RedisGet(cacheKey)
    if err != nil {
        // 缓存未命中，从数据库查询
        mapping, err := GetTeacherSchoolMapping(ctx, teacherId)
        if err != nil {
            return nil, err
        }
        
        // 写入缓存
        mappingJson, _ := json.Marshal(mapping)
        _ = common.RedisSet(cacheKey, string(mappingJson), time.Hour)
        
        return mapping, nil
    }
    
    var mapping TeacherSchoolMapping
    err = json.Unmarshal([]byte(mappingStr), &mapping)
    return &mapping, err
}

// 从数据库获取老师学校映射
func GetTeacherSchoolMapping(ctx context.Context, teacherId string) (*TeacherSchoolMapping, error) {
    var mapping TeacherSchoolMapping
    
    err := DB.Table("teacher_school_mapping tsm").
        Select("tsm.*, s.name as school_name, s.group_name").
        Joins("LEFT JOIN schools s ON tsm.school_id = s.id").
        Where("tsm.teacher_id = ? AND tsm.status = ?", teacherId, MappingStatusActive).
        First(&mapping).Error
    
    if err != nil {
        return nil, err
    }
    
    return &mapping, nil
}

// 批量创建老师映射
func BatchCreateTeacherMappings(mappings []struct {
    TeacherId string `json:"teacher_id"`
    SchoolId  int    `json:"school_id"`
}) (*BatchResult, error) {
    
    result := &BatchResult{
        Total:   len(mappings),
        Success: 0,
        Failed:  0,
        Errors:  make([]string, 0),
    }
    
    return result, DB.Transaction(func(tx *gorm.DB) error {
        for _, mapping := range mappings {
            // 创建映射关系
            mappingRecord := TeacherSchoolMapping{
                TeacherId: mapping.TeacherId,
                SchoolId:  mapping.SchoolId,
                Status:    MappingStatusActive,
                CreatedAt: time.Now().Unix(),
                UpdatedAt: time.Now().Unix(),
            }
            
            if err := tx.Create(&mappingRecord).Error; err != nil {
                result.Failed++
                result.Errors = append(result.Errors, 
                    fmt.Sprintf("创建映射 %s 失败: %v", mapping.TeacherId, err))
                continue
            }
            
            result.Success++
        }
        
        return nil
    })
}

// 清除老师映射缓存
func ClearTeacherMappingCache(teacherId string) error {
    if !common.RedisEnabled {
        return nil
    }
    
    cacheKey := fmt.Sprintf("teacher_mapping:%s", teacherId)
    return common.RedisDel(cacheKey)
}

type BatchResult struct {
    Total   int      `json:"total"`
    Success int      `json:"success"`
    Failed  int      `json:"failed"`
    Errors  []string `json:"errors"`
}
```

```go
// middleware/school_auth.go
package middleware

import (
    "context"
    "errors"
    "fmt"
    "net/http"
    
    "github.com/gin-gonic/gin"
    "github.com/songquanpeng/one-api/common/ctxkey"
    "github.com/songquanpeng/one-api/common/logger"
    "github.com/songquanpeng/one-api/model"
    "gorm.io/gorm"
)

func SchoolAuth() gin.HandlerFunc {
    return func(c *gin.Context) {
        ctx := c.Request.Context()
        
        // 1. 获取老师ID
        teacherId := c.GetHeader("X-User-ID")
        if teacherId == "" {
            // 如果没有老师ID，使用默认用户组（向后兼容）
            logger.Debugf(ctx, "未提供老师ID，使用默认用户组")
            c.Next()
            return
        }
        
        // 2. 查询老师映射（带缓存）
        mapping, err := model.CacheGetTeacherSchoolMapping(ctx, teacherId)
        if err != nil {
            // 3. 如果映射不存在，使用默认分组
            if errors.Is(err, gorm.ErrRecordNotFound) {
                logger.Infof(ctx, "老师 %s 映射不存在，使用默认分组", teacherId)
                
                // 3.1 设置默认分组
                c.Set(ctxkey.Group, "default")
                c.Set(ctxkey.SchoolId, 0)
                c.Set(ctxkey.SchoolName, "默认学校")
                c.Set(ctxkey.TeacherId, teacherId)
                
                // 3.2 创建离线任务
                go model.CreateOfflineTaskForTeacherMapping(teacherId)
                
                c.Next()
                return
            } else {
                logger.Errorf(ctx, "获取老师映射失败: %v", err)
                abortWithMessage(c, http.StatusInternalServerError, "系统错误")
                return
            }
        }
        
        // 4. 验证学校状态
        if mapping.SchoolId == 0 {
            logger.Errorf(ctx, "老师 %s 没有有效的学校映射", teacherId)
            abortWithMessage(c, http.StatusBadRequest, "老师学校映射无效")
            return
        }
        
        // 5. 设置学校上下文
        c.Set(ctxkey.SchoolId, mapping.SchoolId)
        c.Set(ctxkey.SchoolName, mapping.SchoolName)
        c.Set(ctxkey.Group, mapping.GroupName)
        c.Set(ctxkey.TeacherId, teacherId)
        
        logger.Debugf(ctx, "学校认证成功: 老师ID=%s, 学校=%s, 用户组=%s", 
            teacherId, mapping.SchoolName, mapping.GroupName)
        
        c.Next()
    }
}
```

```sql
-- 学校表
CREATE TABLE schools (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL COMMENT '学校名称',
    code VARCHAR(50) NOT NULL UNIQUE COMMENT '学校代码',
    group_name VARCHAR(50) NOT NULL UNIQUE COMMENT '对应的用户组名',

    created_at BIGINT NOT NULL COMMENT '创建时间',
    updated_at BIGINT NOT NULL COMMENT '更新时间',
    INDEX idx_code (code),
    INDEX idx_group_name (group_name),
    INDEX idx_status (status)
);

-- 老师表
CREATE TABLE teachers (
    teacher_id VARCHAR(100) PRIMARY KEY COMMENT '老师ID',
    name VARCHAR(100) DEFAULT '' COMMENT '老师姓名',
    email VARCHAR(200) DEFAULT '' COMMENT '邮箱',
    school_id INT NOT NULL COMMENT '所属学校ID',
    department VARCHAR(100) DEFAULT '' COMMENT '部门',
    status INT DEFAULT 1 COMMENT '状态：1-有效，2-无效',
    created_at BIGINT NOT NULL COMMENT '创建时间',
    updated_at BIGINT NOT NULL COMMENT '更新时间',
    FOREIGN KEY (school_id) REFERENCES schools(id),
    INDEX idx_school_id (school_id),
    INDEX idx_email (email),
    INDEX idx_status (status)
);

-- 老师学校映射表
```
