## 1. 核心设计原则

### 1.1 设计哲学

**核心思想**：OneAPI系统保持完全独立，通过可插拔的身份解析器实现外部身份映射，无任何数据持久化。

```
原始OneAPI：用户ID → 用户组 → 渠道选择
v2.0方案：   外部身份ID → [身份解析器] → 用户组 → 渠道选择
```

### 1.2 零侵入原则

OneAPI系统的原有逻辑完全不变，只通过一个可插拔的身份解析器接口实现外部身份映射。这种设计确保了系统的独立性和纯净性。

#### 2.1.3 课件平台实现

```go
type CoursewareIdentityResolver struct {
    apiClient    *CoursewareAPIClient
    cache        *MemoryCache              // 老师ID -> 用户组的缓存
    groupMapping *IdentityMappingCache     // 抽象化映射缓存
    config       *CoursewareConfig
}

func (c *CoursewareIdentityResolver) ResolveGroup(ctx context.Context, externalIdentity string) string {
    // 1. 检查老师缓存
    if group, hit := c.cache.Get(externalIdentity); hit {
        return group
    }
    
    // 2. 调用课件平台API获取老师信息
    ctx, cancel := context.WithTimeout(ctx, c.config.Timeout)
    defer cancel()
    
    teacherInfo, err := c.apiClient.GetTeacherInfo(ctx, externalIdentity)
    if err != nil {
        // 3. 降级处理 - 返回配置的默认分组
        logger.Warnf(ctx, "课件平台API调用失败，使用默认分组: %v", err)
        c.cache.Set(externalIdentity, c.config.DefaultGroup, c.config.CacheTTL)
        return c.config.DefaultGroup // 配置中指定的默认分组
    }
    
    // 4. 根据学校和学科组信息查找预配置的用户组映射
    var group string
    if teacherInfo.SubjectCode != "" {
        // 优先查找学科组映射
        subjectKey := fmt.Sprintf("school_%s_subject_%s", teacherInfo.SchoolCode, teacherInfo.SubjectCode)
        if mappedGroup := c.groupMapping.GetGroup(subjectKey); mappedGroup != "" {
            group = mappedGroup
            logger.Debugf(ctx, "找到学科组映射: %s -> %s", subjectKey, group)
        }
    }
    
    // 5. 如果没有学科组映射，查找学校映射
    if group == "" {
        schoolKey := fmt.Sprintf("school_%s", teacherInfo.SchoolCode)
        if mappedGroup := c.groupMapping.GetGroup(schoolKey); mappedGroup != "" {
            group = mappedGroup
            logger.Debugf(ctx, "找到学校映射: %s -> %s", schoolKey, group)
        }
    }
    
    // 6. 如果都没有找到，使用默认分组
    if group == "" {
        logger.Warnf(ctx, "未找到老师对应的用户组配置: teacher=%s, school=%s, subject=%s", 
            externalIdentity, teacherInfo.SchoolCode, teacherInfo.SubjectCode)
        group = c.config.DefaultGroup // 配置中指定的默认分组，通常为"default"
    }
    
    // 7. 更新缓存
    c.cache.Set(externalIdentity, group, c.config.CacheTTL)
    
    logger.Infof(ctx, "身份解析成功: teacher=%s, group=%s, school=%s, subject=%s", 
        externalIdentity, group, teacherInfo.SchoolCode, teacherInfo.SubjectCode)
    return group
}
```

```go
type CoursewareAPIClient struct {
    baseURL    string        // 课件平台API地址
    apiKey     string        // 认证密钥
    httpClient *http.Client  // HTTP客户端
}

func (c *CoursewareAPIClient) GetTeacherInfo(ctx context.Context, teacherId string) (*TeacherInfo, error) {
    // 构建请求URL
    url := fmt.Sprintf("%s/teacher/%s/info", c.baseURL, teacherId)
    
    // 创建HTTP请求
    req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
    if err != nil {
        return nil, err
    }
    
    // 设置认证头
    req.Header.Set("Authorization", "Bearer "+c.apiKey)
    req.Header.Set("Content-Type", "application/json")
    
    // 发送请求
    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    // 解析响应
    var response TeacherInfoResponse
    err = json.NewDecoder(resp.Body).Decode(&response)
    if err != nil {
        return nil, err
    }
    
    if !response.Success {
        return nil, fmt.Errorf("API返回错误: %s", response.Message)
    }
    
    return response.Data, nil
}
```

#### 2.3.3 完整的ResolveGroup实现

```go
// OneAPI团队实现：完整的身份解析逻辑（查找预配置映射）
func (c *CoursewareIdentityResolver) ResolveGroup(ctx context.Context, externalIdentity string) string {
    // 1. 先检查老师缓存
    if group, hit := c.cache.Get(externalIdentity); hit {
        logger.Debugf(ctx, "缓存命中: teacher=%s, group=%s", externalIdentity, group)
        return group
    }
    
    // 2. 超时控制
    ctx, cancel := context.WithTimeout(ctx, c.config.Timeout)
    defer cancel()
    
    // 3. 缓存未命中，调用课件平台API获取老师信息
    teacherInfo, err := c.apiClient.GetTeacherInfo(ctx, externalIdentity)
    if err != nil {
        // 4. 错误分类处理
        switch {
        case ctx.Err() == context.DeadlineExceeded:
            logger.Warnf(ctx, "API调用超时: teacher=%s", externalIdentity)
        case strings.Contains(err.Error(), "connection refused"):
            logger.Warnf(ctx, "课件平台不可用: teacher=%s", externalIdentity)
        default:
            logger.Warnf(ctx, "API调用失败: teacher=%s, error=%v", externalIdentity, err)
        }
        
        // 5. 降级到配置的默认分组，确保系统可用
        c.cache.Set(externalIdentity, c.config.DefaultGroup, c.config.CacheTTL)
        return c.config.DefaultGroup // 配置中指定的默认分组
    }
    
    // 6. 根据老师信息查找预配置的用户组映射（不是动态生成！）
    group := c.findMappedGroup(teacherInfo)
    
    // 7. 如果没有找到映射，使用默认分组
    if group == "" {
        logger.Warnf(ctx, "未找到老师对应的用户组配置: teacher=%s, school=%s, subject=%s", 
            externalIdentity, teacherInfo.SchoolCode, teacherInfo.SubjectCode)
        group = c.config.DefaultGroup
    }
    
    // 8. 更新缓存
    c.cache.Set(externalIdentity, group, c.config.CacheTTL)
    
    logger.Infof(ctx, "身份解析成功: teacher=%s, group=%s, school=%s, subject=%s", 
        externalIdentity, group, teacherInfo.SchoolCode, teacherInfo.SubjectCode)
    return group
}
```

#### 2.4.3 抽象化映射缓存实现

```go
// 抽象化映射缓存 - 通用键值映射，支持任意业务场景
type IdentityMappingCache struct {
    mappings map[string]string // external_key -> oneapi_group
    metadata map[string]map[string]interface{} // external_key -> metadata
    mutex    sync.RWMutex
    db       *gorm.DB          // 数据库连接
}

// 抽象化映射记录结构
type IdentityMapping struct {
    ID          int                    `gorm:"primaryKey" json:"id"`
    ExternalKey string                 `gorm:"uniqueIndex;not null" json:"external_key"`
    OneAPIGroup string                 `gorm:"not null" json:"oneapi_group"`
    Metadata    datatypes.JSON         `json:"metadata,omitempty"`       // JSON字段，存储任意业务属性

    Priority    int                    `gorm:"default:100" json:"priority"`       // 优先级
    Enabled     bool                   `gorm:"default:true" json:"enabled"`
    CreatedAt   time.Time              `json:"created_at"`
    UpdatedAt   time.Time              `json:"updated_at"`
    CreatedBy   string                 `gorm:"default:'admin'" json:"created_by"`
    Description string                 `json:"description,omitempty"`
}

func NewIdentityMappingCache(db *gorm.DB) *IdentityMappingCache {
    cache := &IdentityMappingCache{
        mappings: make(map[string]string),
        metadata: make(map[string]map[string]interface{}),
        db:       db,
    }
    
    // 初始化数据库表
    cache.initTable()
    
    // 加载映射到内存
    cache.loadFromDB()
    
    return cache
}

// 从数据库加载所有映射到内存
func (c *IdentityMappingCache) loadFromDB() error {
    var mappings []IdentityMapping
    err := c.db.Where("enabled = ?", true).Order("priority ASC, created_at ASC").Find(&mappings).Error
    if err != nil {
        return err
    }
    
    c.mutex.Lock()
    defer c.mutex.Unlock()
    
    // 清空并重新加载
    c.mappings = make(map[string]string)
    c.metadata = make(map[string]map[string]interface{})
    
    for _, mapping := range mappings {
        c.mappings[mapping.ExternalKey] = mapping.OneAPIGroup
        
        // 解析JSON元数据
        if mapping.Metadata != nil {
            var meta map[string]interface{}
            if err := json.Unmarshal(mapping.Metadata, &meta); err == nil {
                c.metadata[mapping.ExternalKey] = meta
            }
        }
    }
    
    logger.Infof(context.Background(), "身份映射缓存加载完成，共 %d 条记录", len(c.mappings))
    return nil
}

// 获取用户组 (线程安全)
func (c *IdentityMappingCache) GetGroup(key string) string {
    c.mutex.RLock()
    defer c.mutex.RUnlock()
    
    return c.mappings[key]
}

// 获取元数据 (线程安全)
func (c *IdentityMappingCache) GetMetadata(key string) map[string]interface{} {
    c.mutex.RLock()
    defer c.mutex.RUnlock()
    
    return c.metadata[key]
}

// 获取显示名称 (从元数据中提取)
func (c *IdentityMappingCache) GetDisplayName(key string) string {
    if meta := c.GetMetadata(key); meta != nil {
        if displayName, ok := meta["display_name"].(string); ok {
            return displayName
        }
    }
    return key // fallback到原始key
}

// 按分类获取映射列表
func (c *IdentityMappingCache) GetMappingsByCategory(category string) map[string]string {
    c.mutex.RLock()
    defer c.mutex.RUnlock()
    
    result := make(map[string]string)
    for key, group := range c.mappings {
        if meta := c.metadata[key]; meta != nil {
            if metaCategory, ok := meta["category"].(string); ok && metaCategory == category {
                result[key] = group
            }
        }
    }
    return result
}

// 初始化数据库表
func (c *IdentityMappingCache) initTable() error {
    return c.db.AutoMigrate(&IdentityMapping{})
}

// 强制刷新缓存 (用于API修改后)
func (c *IdentityMappingCache) Refresh() error {
    return c.loadFromDB()
}
```

```go
// 7. 批量导入映射 - POST /api/v1/admin/identity-mappings/batch
func (api *IdentityMappingAPI) BatchImport(c *gin.Context) {
    var req struct {
        Mappings []struct {
            ExternalKey string                 `json:"external_key"`
            OneAPIGroup string                 `json:"oneapi_group"`
            Metadata    map[string]interface{} `json:"metadata"`
            Category    string                 `json:"category"`
            Priority    int                    `json:"priority"`
            Description string                 `json:"description"`
        } `json:"mappings"`
        OverwriteExisting bool `json:"overwrite_existing"` // 是否覆盖已存在的映射
    }
    
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": "参数错误: " + err.Error()})
        return
    }
    
    var results []map[string]interface{}
    successCount := 0
    
    for i, mappingReq := range req.Mappings {
        result := map[string]interface{}{
            "index":        i,
            "external_key": mappingReq.ExternalKey,
            "success":      false,
        }
        
        // 验证必填字段
        if mappingReq.ExternalKey == "" || mappingReq.OneAPIGroup == "" {
            result["error"] = "external_key和oneapi_group不能为空"
            results = append(results, result)
            continue
        }
        
        // 检查用户组是否存在
        if !api.validateUserGroup(mappingReq.OneAPIGroup) {
            result["error"] = "OneAPI用户组不存在: " + mappingReq.OneAPIGroup
            results = append(results, result)
            continue
        }
        
        // 序列化元数据
        var metadataJSON datatypes.JSON
        if mappingReq.Metadata != nil {
            metadataBytes, _ := json.Marshal(mappingReq.Metadata)
            metadataJSON = datatypes.JSON(metadataBytes)
        }
        
        // 设置默认值
        if mappingReq.Category == "" {
            mappingReq.Category = "default"
        }
        if mappingReq.Priority == 0 {
            mappingReq.Priority = 100
        }
        
        mapping := IdentityMapping{
            ExternalKey: mappingReq.ExternalKey,
            OneAPIGroup: mappingReq.OneAPIGroup,
            Metadata:    metadataJSON,
            Category:    mappingReq.Category,
            Priority:    mappingReq.Priority,
            Description: mappingReq.Description,
            Enabled:     true,
            CreatedBy:   c.GetString("username"),
        }
        
        // 处理重复键
        if req.OverwriteExisting {
            err := api.db.Where("external_key = ?", mappingReq.ExternalKey).Assign(mapping).FirstOrCreate(&mapping).Error
            if err != nil {
                result["error"] = err.Error()
            } else {
                result["success"] = true
                successCount++
            }
        } else {
            err := api.db.Create(&mapping).Error
            if err != nil {
                if strings.Contains(err.Error(), "Duplicate entry") {
                    result["error"] = "外部键已存在"
                } else {
                    result["error"] = err.Error()
                }
            } else {
                result["success"] = true
                successCount++
            }
        }
        
        results = append(results, result)
    }
    
    // 刷新缓存
    if successCount > 0 {
        api.cache.Refresh()
    }
    
    c.JSON(http.StatusOK, gin.H{
        "message":       fmt.Sprintf("批量导入完成，成功 %d 条，失败 %d 条", successCount, len(req.Mappings)-successCount),
        "success_count": successCount,
        "total_count":   len(req.Mappings),
        "results":       results,
    })
}
```

### 2.5 内存缓存实现

```go
// 简单的内存缓存实现
type MemoryCache struct {
    data   map[string]CacheEntry
    mutex  sync.RWMutex
    maxSize int
}

type CacheEntry struct {
    Value     string    // 用户组名
    ExpiresAt time.Time // 过期时间
}

func NewMemoryCache(maxSize int) *MemoryCache {
    cache := &MemoryCache{
        data:    make(map[string]CacheEntry),
        maxSize: maxSize,
    }
    
    // 启动清理协程
    go cache.startCleanup()
    
    return cache
}

func (m *MemoryCache) Get(key string) (string, bool) {
    m.mutex.RLock()
    defer m.mutex.RUnlock()
    
    entry, exists := m.data[key]
    if !exists {
        return "", false
    }
    
    if time.Now().After(entry.ExpiresAt) {
        return "", false
    }
    
    return entry.Value, true
}

func (m *MemoryCache) Set(key, value string, ttl time.Duration) {
    m.mutex.Lock()
    defer m.mutex.Unlock()
    
    // 如果缓存满了，清理过期条目
    if len(m.data) >= m.maxSize {
        m.cleanExpired()
    }
    
    // 如果还是满的，随机删除一些条目
    if len(m.data) >= m.maxSize {
        for k := range m.data {
            delete(m.data, k)
            if len(m.data) < m.maxSize*8/10 { // 保持80%容量
                break
            }
        }
    }
    
    m.data[key] = CacheEntry{
        Value:     value,
        ExpiresAt: time.Now().Add(ttl),
    }
}
```
